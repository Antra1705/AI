================================================================
Damping characterisation: servo locked at each stiffness, 4 cars per cell, PWM 200.
smooth/gravel cost = RMS body accel (m/s^2) + 3 x bottom-outs per metre
breaker/pothole cost = mean peak body accel / 10 + bottom-outs per event

surface    s=0.0 ( 45 deg)     s=0.2 ( 66 deg)     s=0.4 ( 87 deg)     s=0.6 (108 deg)     s=0.8 (129 deg)     s=1.0 (150 deg)  
smooth     0.085                 0.057                 0.051                 0.049                 0.049                 0.049              
gravel     1.170                 2.178                 3.267                 4.355                 5.419                 6.447              
breaker    8.432                 4.285                 2.223                 1.137                 0.687                 0.590              
pothole    17.832                9.090                 6.231                 5.158                 3.616                 2.847

constexpr float STIFF_SMOOTH  = 0.8f;
